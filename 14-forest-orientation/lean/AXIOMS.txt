@FO.round_gain : ∀ {n : Nat} {D : FO.Pos n} {x y : Fin n} {ws : List (Fin n)},
  FO.Forest D →
    FO.Valid D →
      ¬FO.Conn D x y →
        FO.NPath D x ws →
          ws.length = FO.hh D x →
            2 * FO.hh D x + FO.eps D x ≤ 2 * FO.hh D y + FO.eps D y →
              FO.Phi D + 3 * ↑ws.length ≤ FO.Phi (FO.nreply D x y ws)
@FO.ruleL_round : ∀ {n : Nat} {D : FO.Pos n},
  FO.Forest D →
    FO.Valid D →
      ∀ {u v : Fin n},
        ¬FO.Conn D u v →
          FO.ruleL D u v ∈ FO.replies D u v ∧
            FO.Forest (FO.ruleL D u v) ∧
              FO.Valid (FO.ruleL D u v) ∧
                List.length (FO.ruleL D u v) = List.length D + 1 ∧
                  FO.Phi D + 3 * ↑(FO.cost D (FO.ruleL D u v)) ≤ FO.Phi (FO.ruleL D u v)
