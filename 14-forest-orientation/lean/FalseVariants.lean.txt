-- the Type Lemma with "the index always goes up" (false: an R_k pair may stay R_k at cost k+1)
example {n : Nat} {D : Pos n} {t hd : Fin n} {ws : List (Fin n)} (hF : Forest D)
    (hc : ¬ Conn D t hd) (hp : NPath D t ws) {i : Nat} (ht : HasIdx D i t)
    (hh : HasIdx D i hd) : HasIdx (nreply D t hd ws) (i + 1) t := by
  rcases lemmaB hF hc hp ht hh.1 with ⟨h1, h2⟩ | ⟨h1, h2⟩
  · exact TypeGe.mono h2 (by omega) (by omega)
  · exact TypeGe.mono h2 (by omega) (by omega)
-- the final theorem with constant 4 in place of 3 from the same forcing statement
